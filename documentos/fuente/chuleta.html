SELECT con sintaxis estricta

SELECT FROM ekko AS h
  INNER JOIN ekpo AS p ON p~ebeln = h~ebeln
  FIELDS h~ebeln, h~lifnr, p~ebelp, p~matnr, p~netwr
  WHERE h~bukrs IN @s_bukrs
    AND p~loekz = @abap_false
  ORDER BY h~ebeln, p~ebelp
  INTO TABLE @DATA(lt_po)
  UP TO 500 ROWS.
Comas entre campos + @ en variables ABAP = sintaxis estricta. Necesaria para expresiones, declaración en línea y ABAP Cloud.

Behavior definition (gestionado con draft)

managed implementation in class zbp_r_order unique;
strict ( 2 );
with draft;
define behavior for ZR_Order alias Order
persistent table zorder
draft table zorder_d
lock master total etag LastChangedAt
authorization master ( instance )
etag master LocalLastChangedAt
{
  create; update; delete;
  field ( readonly, numbering : managed ) OrderUUID;
  field ( mandatory ) Customer;
  action ( features : instance ) approve result [1] $self;
  determination setStatus on modify { create; }
  validation checkCustomer on save
    { create; field Customer; }
  draft action Edit; draft action Activate optimized;
  draft action Discard; draft action Resume;
  draft determine action Prepare
    { validation checkCustomer; }
}

EML: crear, modificar, borrar, ejecutar

MODIFY ENTITIES OF zr_order
  ENTITY Order
    CREATE FIELDS ( Customer Description )
    WITH VALUE #( ( %cid        = 'O1'
                    Customer    = '1000'
                    Description = `Nuevo` ) )
    CREATE BY \_Item FIELDS ( Material Quantity )
    WITH VALUE #( ( %cid_ref = 'O1'
                    %target  = VALUE #( ( %cid     = 'I1'
                                          Material = 'M-01'
                                          Quantity = 5 ) ) ) )
  MAPPED DATA(ls_mapped)
  FAILED DATA(ls_failed)
  REPORTED DATA(ls_reported).

MODIFY ENTITIES OF zr_order ENTITY Order
  UPDATE FIELDS ( Status )
  WITH VALUE #( ( %tky = ls-%tky Status = 'A' ) ).

MODIFY ENTITIES OF zr_order ENTITY Order
  DELETE FROM VALUE #( ( %tky = ls-%tky ) ).

MODIFY ENTITIES OF zr_order ENTITY Order
  EXECUTE approve FROM VALUE #( ( %tky = ls-%tky ) ).
